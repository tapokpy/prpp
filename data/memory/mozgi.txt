Вот безопасный и проверенный способ обновить pip, системные инструменты и все зависимости проекта внутри вашего виртуального окружения.

# 1. Переходим в папку проекта
cd /opt/prpp

# 2. Активируем виртуальное окружение
source .venv/bin/activate

# 3. Обновляем pip, setuptools и wheel
# 🔧 ВАЖНО: setuptools<82 нужен из-за конфликта с torch 2.12.0
pip install --upgrade pip "setuptools<82" wheel

# 4. Обновляем зависимости проекта
# --upgrade-strategy eager заставит pip обновить даже те пакеты,
# которые уже установлены, но имеют более новые совместимые версии
pip install -r requirements.txt --upgrade

# 5. Деактивируем окружение
deactivate
